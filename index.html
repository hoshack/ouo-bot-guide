How it Works

How are views registered?

Your bot must simulate exactly the steps a browser takes for a view to be valid.

1

Get ouo.io Link

Bot receives the link from the user. For example: https://ouo.io/AbCdEf

💡 Link can be sent via Telegram, Discord, or any platform.
2

First Request — Ad Page

Send a GET request to the ouo.io link. The server returns an HTML page with a hidden form and token.

# Get first page and extract token
GET https://ouo.io/AbCdEf
# Response: HTML with form and _csrf token
3

Extract CSRF Token

Parse the _csrf field value from the received HTML. This token is required for the next step.

# Extract from HTML
<input type="hidden" name="_csrf" value="TOKEN_HERE">
4

Send POST — Register View

Send a POST request with the CSRF token. This officially registers the view and returns the final link.

# Submit form and register view
POST https://ouo.io/go/AbCdEf
Body: _csrf=TOKEN&...
5

Get and Send Final Link

The final link is extracted from the Location header or response body and sent to the user.

✅ View registered + User got the link — everyone wins!

import requests
from bs4 import BeautifulSoup
import re

def bypass_ouo(short_url: str) -> str:
    """
    Receives ouo.io link, registers view 
    and returns final link.
    """
    session = requests.Session()
    
    # Browser-like headers
    headers = {
        "User-Agent": "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36",
        "Accept": "text/html,application/xhtml+xml,application/xml;q=0.9,*/*;q=0.8",
        "Accept-Language": "en-US,en;q=0.5",
        "Referer": "https://ouo.io/",
    }
    
    # Step 1: Get ad page
    response = session.get(short_url, headers=headers, allow_redirects=True)
    
    # Step 2: Extract CSRF token
    soup = BeautifulSoup(response.text, "html.parser")
    csrf_token = soup.find("input", {"name": "_csrf"})
    
    if not csrf_token:
        raise ValueError("CSRF token not found")
    
    # Step 3: Register view + get final link
    link_id = short_url.split("/")[-1]
    post_url = f"https://ouo.io/go/{link_id}"
    
    payload = {
        "_csrf": csrf_token["value"],
    }
    
    final = session.post(
        post_url,
        data=payload,
        headers=headers,
        allow_redirects=True
    )
    
    return final.url  # Final link

const axios = require('axios');
const cheerio = require('cheerio');
const { Telegraf } = require('telegraf');

async function bypassOuo(shortUrl) {
  const headers = {
    'User-Agent': 'Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36',
    'Accept': 'text/html,application/xhtml+xml,application/xml;q=0.9',
    'Referer': 'https://ouo.io/',
  };

  // Step 1: Get page and cookies
  const firstRes = await axios.get(shortUrl, {
    headers,
    maxRedirects: 10,
    withCredentials: true,
  });

  // Step 2: Extract CSRF token
  const $ = cheerio.load(firstRes.data);
  const csrfToken = $('input[name="_csrf"]').val();

  if (!csrfToken) throw new Error('CSRF token not found');

  // Step 3: Register view
  const linkId = shortUrl.split('/').pop();
  const postUrl = `https://ouo.io/go/${linkId}`;

  const params = new URLSearchParams();
  params.append('_csrf', csrfToken);

  const finalRes = await axios.post(postUrl, params.toString(), {
    headers: {
      ...headers,
      'Content-Type': 'application/x-www-form-urlencoded',
    },
    maxRedirects: 10,
  });

  return finalRes.request.res.responseUrl;
}

<?php

function bypassOuo(string $shortUrl): string {
    $ch = curl_init();
    
    $headers = [
        'User-Agent: Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36',
        'Accept: text/html,application/xhtml+xml,application/xml;q=0.9',
        'Referer: https://ouo.io/',
    ];
    
    // Step 1: Get page
    curl_setopt_array($ch, [
        CURLOPT_URL => $shortUrl,
        CURLOPT_RETURNTRANSFER => true,
        CURLOPT_FOLLOWLOCATION => true,
        CURLOPT_HTTPHEADER => $headers,
        CURLOPT_COOKIEJAR => '/tmp/cookies.txt',
        CURLOPT_COOKIEFILE => '/tmp/cookies.txt',
        CURLOPT_SSL_VERIFYPEER => false,
    ]);
    
    $html = curl_exec($ch);
    
    // Step 2: Extract CSRF token
    preg_match('/name="_csrf"\s+value="([^"]+)"/', $html, $matches);
    
    if (empty($matches[1])) {
        throw new Exception('CSRF token not found');
    }
    
    $csrf = $matches[1];
    $linkId = basename($shortUrl);
    
    // Step 3: Send POST and register view
    curl_setopt_array($ch, [
        CURLOPT_URL => "https://ouo.io/go/{$linkId}",
        CURLOPT_POST => true,
        CURLOPT_POSTFIELDS => http_build_query(['_csrf' => $csrf]),
        CURLOPT_FOLLOWLOCATION => true,
    ]);
    
    curl_exec($ch);
    $finalUrl = curl_getinfo($ch, CURLINFO_EFFECTIVE_URL);
    curl_close($ch);
    
    return $finalUrl;
}
